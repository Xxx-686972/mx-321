② JS：mx_buildSwiper4 + creative 电影墙 + 按钮切换 
// 保存全局实例，供切换时销毁重建
var mx_swiper4 = null;

// 核心函数：销毁旧实例 → 按新 effect 重建
function mx_buildSwiper4(mx_fx) {
  // 关键：换 effect 必须先 destroy，否则新配置不生效
  if (mx_swiper4) {
    mx_swiper4.destroy(true, true);
  }

  var mx_opts = {
    loop: true,
    effect: mx_fx,          // ← 这里切换效果！
    speed: 800,
    autoplay: { delay: 3200, disableOnInteraction: false },
    pagination: { el: '#wyPag4', clickable: true },
    keyboard: { enabled: true }  // 超纲：方向键翻页
  };

  // ===== creative 电影墙自定义参数 =====
  if (mx_fx === 'creative') {
    mx_opts.creativeEffect = {
      limitProgress: 3,
      // prev：上一张 → 缩到 0.7 + 左移 -60% + 旋转 -18° + 压进深 -400
      prev: {
        shadow: true,
        translate: ['-60%', 0, -400],
        rotate: [0, 0, -18],
        scale: 0.7
      },
      // next：下一张 → 放大 1.15 + 右移 60% + 旋转 18°
      next: {
        shadow: true,
        translate: ['60%', 0, -400],
        rotate: [0, 0, 18],
        scale: 1.15
      }
    };
  }

  mx_swiper4 = new Swiper('#wySw4', mx_opts);
}

// 初始化：默认 slide 平推
mx_buildSwiper4('slide');

// 按钮点击切换：遍历所有 .wy-fx button
document.querySelectorAll('.wy-fx button').forEach(function(btn) {
  btn.addEventListener('click', function() {
    // 1. 高亮当前按钮（加 wy-on，其余移除）
    document.querySelectorAll('.wy-fx button').forEach(function(mx_btn) {
      mx_btn.classList.remove('wy-on');
    });
    btn.classList.add('wy-on');

    // 2. 销毁 + 重建（data-fx 存的就是目标效果名）
    mx_buildSwiper4(btn.dataset.fx);
  });
});
